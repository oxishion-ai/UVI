Les methodes speciales permettent de personnaliser le comportement d'une classe.

class Vecteur:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __add__(self, autre):
        return Vecteur(self.x + autre.x, self.y + autre.y)

    def __repr__(self):
        return "Vecteur(" + str(self.x) + ", " + str(self.y) + ")"

    def __eq__(self, autre):
        return self.x == autre.x and self.y == autre.y

Un generateur utilise le mot-cle yield au lieu de return, et produit ses valeurs une par une, a la demande.

def compteur(debut, fin):
    valeur = debut
    while valeur <= fin:
        yield valeur
        valeur += 1

def fibonacci():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

Un decorateur est une fonction qui modifie le comportement d'une autre fonction sans changer son code.

def chronometre(fonction):
    import time
    def enveloppe(*args, **kwargs):
        debut = time.time()
        resultat = fonction(*args, **kwargs)
        duree = time.time() - debut
        print("temps ecoule :", duree)
        return resultat
    return enveloppe

@chronometre
def calcul_long():
    total = 0
    for i in range(1000000):
        total += i
    return total

Un gestionnaire de contexte utilise le mot-cle with, et repose sur les methodes __enter__ et __exit__.

class Fichier:

        self.fichier = open(self.chemin, self.mode)
        return self.fichier

    def __exit__(self, type_exc, valeur_exc, trace):

La gestion des exceptions utilise try, except, else, finally.

Un algorithme de tri rapide (quicksort) recursif.

def tri_rapide(tableau):
    if len(tableau) <= 1:
        return tableau
    pivot = tableau[len(tableau) // 2]
    gauche = [x for x in tableau if x < pivot]
    milieu = [x for x in tableau if x == pivot]
    droite = [x for x in tableau if x > pivot]
    return tri_rapide(gauche) + milieu + tri_rapide(droite)

Une recherche binaire dans un tableau trie.

def recherche_binaire(tableau, cible):
    gauche, droite = 0, len(tableau) - 1
    while gauche <= droite:
        milieu = (gauche + droite) // 2
        if tableau[milieu] == cible:
            return milieu
        elif tableau[milieu] < cible:
            gauche = milieu + 1
        else:
            droite = milieu - 1
    return -1

Un arbre binaire de recherche simple.

class Noeud:
    def __init__(self, valeur):
        self.valeur = valeur
        self.gauche = None
        self.droite = None

class ArbreBinaire:
    def __init__(self):
        self.racine = None

    def inserer(self, valeur):
        if self.racine is None:
            self.racine = Noeud(valeur)
        else:
            self._inserer_recursif(self.racine, valeur)

    def _inserer_recursif(self, noeud, valeur):
        if valeur < noeud.valeur:
            if noeud.gauche is None:
                noeud.gauche = Noeud(valeur)
            else:
                self._inserer_recursif(noeud.gauche, valeur)
        else:
            if noeud.droite is None:
                noeud.droite = Noeud(valeur)
            else:
                self._inserer_recursif(noeud.droite, valeur)

Un exemple complet : une fonction qui compte les mots d'un texte.

def compter_mots(texte):
    mots = texte.lower().split()
    compteur = {}
    for mot in mots:
        compteur[mot] = compteur.get(mot, 0) + 1
    return compteur

Un exemple complet : une classe qui represente une pile de cartes.

class Carte:
    def __init__(self, valeur, couleur):
        self.valeur = valeur
        self.couleur = couleur

    def __repr__(self):
        return str(self.valeur) + " de " + self.couleur

class Jeu:
    def __init__(self):
        valeurs = range(1, 14)
        couleurs = ["coeur", "carreau", "trefle", "pique"]
        self.cartes = [Carte(v, c) for v in valeurs for c in couleurs]

    def melanger(self):
        import random
        random.shuffle(self.cartes)

    def piocher(self):
        return self.cartes.pop()
